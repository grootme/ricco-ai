   263→   263→    async def search(
   264→   264→        self, 
   265→   265→        query: str, 
   266→   266→        limit: int = 10,
   267→   267→        filters: Optional[Dict[str, Any]] = None
   268→   268→    ) -> List[Memory]:
   269→   269→        """Buscar memorias por contenido (búsqueda text simple)"""
   270→   270→        await self._init_db()
   271→   271→        
   272→   272→        conn = sqlite3.connect(self.db_path)
   273→   273→        cursor = conn.cursor()
   274→   274→        
   275→   275→        sql = """
   276→   276→            SELECT id, agent_id, type, category, key, content, embedding,
   277→   277→                   importance, access_count, last_accessed_at, expires_at,
   278→   278→                   session_id, source_type, source_id, created_at, updated_at
   279→   279→            FROM memories WHERE content LIKE ?
   280→   280→        """
   281→   281→        params = [f"%{query}%"]
   282→   282→        
   283→   283→        if filters:
   284→   284→            if "agent_id" in filters:
   285→   285→                sql += " AND agent_id = ?"
   286→   286→                params.append(str(filters["agent_id"]))
   287→   287→            if "type" in filters:
   288→   288→                sql += " AND type = ?"
   289→   289→                params.append(filters["type"].value if hasattr(filters["type"], "value") else filters["type"])
   290→   290→            if "session_id" in filters:
   291→   291→                sql += " AND session_id = ?"
   292→   292→                params.append(str(filters["session_id"]))
   293→   293→        
   294→   294→        sql += f" ORDER BY importance DESC, created_at DESC LIMIT {limit}"
   295→   295→        
   296→   296→        cursor.execute(sql, params)
   297→   297→        rows = cursor.fetchall()
   298→   298→        conn.close()
   299→   299→        
   300→   300→        return [self._row_to_memory(row) for row in rows]

   317→   317→    async def update(self, memory_id: UUID, data: Dict[str, Any]) -> bool:
   318→   318→        """Actualizar una memoria"""
   319→   319→        await self._init_db()
   320→   320→        
   321→   321→        if not data:
   322→   322→            return False
   323→   323→        
   324→   324→        set_clauses = []
   325→   325→        params = []
   326→   326→        
   327→   327→        for key, value in data.items():
   328→   328→            if key in ["content", "importance", "access_count", "last_accessed_at", "expires_at"]:
   329→   329→                set_clauses.append(f"{key} = ?")
   330→   330→                if isinstance(value, datetime):
   331→   331→                    params.append(value.isoformat())
   332→   332→                else:
   333→   333→                    params.append(value)
   334→   334→        
   335→   335→        if not set_clauses:
   336→   336→            return False
   337→   337→        
   338→   338→        set_clauses.append("updated_at = ?")
   339→   339→        params.append(datetime.utcnow().isoformat())
   340→   340→        
   341→   341→        params.append(str(memory_id))
   342→   342→        
   343→   343→        conn = sqlite3.connect(self.db_path)
   344→   344→        cursor = conn.cursor()
   345→   345→        
   346→   346→        sql = f"UPDATE memories SET {', '.join(set_clauses)} WHERE id = ?"
   347→   347→        cursor.execute(sql, params)
   348→   348→        updated = cursor.rowcount > 0
   349→   349→        
   350→   350→        conn.commit()
   351→   351→        conn.close()
   352→   352→        
   353→   353→        return updated

   428→   428→# VECTOR MEMORY STORE (Semantic/Long-term)
   429→   429→# ============================================
   430→   430→
   431→   431→class VectorMemoryStore(MemoryStore):
   432→   432→    """
   433→   433→    Almacén de memoria vectorial para búsqueda semántica.
   434→   434→    
   435→   435→    Usa Milvus para almacenamiento de embeddings y búsqueda vectorial.
   436→   436→    """
   437→   437→    
   438→   438→    def __init__(
   439→   439→        self,
   440→   440→        host: str = "localhost",
   441→   441→        port: int = 19530,
   442→   442→        collection_name: str = "cognitive_memory",
   443→   443→        embedding_provider: Optional[EmbeddingProvider] = None
   444→   444→    ):
   445→   445→        self.host = host
   446→   446→        self.port = port
   447→   447→        self.collection_name = collection_name
   448→   448→        self.embedding_provider = embedding_provider
   449→   449→        self._collection = None
   450→   450→    
   451→   451→    async def _get_collection(self):
   452→   452→        """Obtener o crear colección en Milvus"""
   453→   453→        if self._collection is not None:
   454→   454→            return self._collection
   455→   455→        
   456→   456→        try:
   457→   457→            from pymilvus import Collection, connections, FieldSchema, CollectionSchema, DataType
   458→   458→            
   459→   459→            connections.connect("default", host=self.host, port=self.port)
   460→   460→            
   461→   461→            fields = [
   462→   462→                FieldSchema(name="id", dtype=DataType.VARCHAR, max_length=36, is_primary=True),
   463→   463→                FieldSchema(name="agent_id", dtype=DataType.VARCHAR, max_length=36),
   464→   464→                FieldSchema(name="type", dtype=DataType.VARCHAR, max_length=20),
   465→   465→                FieldSchema(name="content", dtype=DataType.VARCHAR, max_length=65535),
   466→   466→                FieldSchema(name="embedding", dtype=DataType.FLOAT_VECTOR, dim=1536),
   467→   467→                FieldSchema(name="importance", dtype=DataType.FLOAT),
   468→   468→                FieldSchema(name="created_at", dtype=DataType.INT64),
   469→   469→            ]
   470→   470→            
   471→   471→            schema = CollectionSchema(fields=fields, description="Cognitive Memory")
   472→   472→            
   473→   473→            self._collection = Collection(self.collection_name, schema)
   474→   474→            
   475→   475→            # Crear índice para búsqueda vectorial
   476→   476→            index_params = {
   477→   477→                "metric_type": "COSINE",
   478→   478→                "index_type": "IVF_FLAT",
   479→   479→                "params": {"nlist": 1024}
   480→   480→            }
   481→   481→            self._collection.create_index("embedding", index_params)
   482→   482→            
   483→   483→            return self._collection
   484→   484→            
   485→   485→        except ImportError:
   486→   486→            raise RuntimeError("pymilvus is required for VectorMemoryStore")
   487→   487→        except Exception as e:
   488→   488→            raise RuntimeError(f"Failed to connect to Milvus: {e}")
   489→   489→    
   490→   490→    async def store(self, memory: Memory) -> UUID:
   491→   491→        """Almacenar una memoria con embedding"""
   492→   492→        collection = await self._get_collection()
   493→   493→        
   494→   494→        # Generar embedding si no existe
   495→   495→        if not memory.embedding and self.embedding_provider:
   496→   496→            memory.embedding = await self.embedding_provider.embed(memory.content)
   497→   497→        
   498→   498→        if not memory.embedding:
   499→   499→            raise ValueError("Memory must have embedding for vector store")
   500→   500→        
   501→   501→        data = [
   502→   502→            [str(memory.id)],
   503→   503→            [str(memory.agent_id)],
   504→   504→            [memory.type.value],
   505→   505→            [memory.content],
   506→   506→            [memory.embedding],
   507→   507→            [memory.importance],
   508→   508→            [int(memory.created_at.timestamp())]
   509→   509→        ]
   510→   510→        
   511→   511→        collection.insert(data)
   512→   512→        collection.flush()
   513→   513→        
   514→   514→        return memory.id
   515→   515→    
   516→   516→    async def retrieve(self, memory_id: UUID) -> Optional[Memory]:
   517→   517→        """Recuperar una memoria por ID"""
   518→   518→        collection = await self._get_collection()
   519→   519→        
   520→   520→        collection.load()
   521→   521→        results = collection.query(
   522→   522→            expr=f'id == "{str(memory_id)}"',
   523→   523→            output_fields=["id", "agent_id", "type", "content", "embedding", "importance", "created_at"]
   524→   524→        )
   525→   525→        
   526→   526→        if not results:
   527→   527→            return None
   528→   528→        
   529→   529→        return self._result_to_memory(results[0])
   530→   530→    
   531→   531→    async def search(
   532→   532→        self, 
   533→   533→        query: str, 
   534→   534→        limit: int = 10,
   535→   535→        filters: Optional[Dict[str, Any]] = None
   536→   536→    ) -> List[Memory]:
   537→   537→        """Búsqueda semántica por similitud vectorial"""
   538→   538→        collection = await self._get_collection()
   539→   539→        
   540→   540→        if not self.embedding_provider:
   541→   541→            raise ValueError("Embedding provider required for semantic search")
   542→   542→        
   543→   543→        # Generar embedding de la query
   544→   544→        query_embedding = await self.embedding_provider.embed(query)
   545→   545→        
   546→   546→        # Construir filtro
   547→   547→        filter_expr = None
   548→   548→        if filters:
   549→   549→            conditions = []
   550→   550→            if "agent_id" in filters:
   551→   551→                conditions.append(f'agent_id == "{str(filters["agent_id"])}"')
   552→   552→            if "type" in filters:
   553→   553→                type_val = filters["type"].value if hasattr(filters["type"], "value") else filters["type"]
   554→   554→                conditions.append(f'type == "{type_val}"')
   555→   555→            if conditions:
   556→   556→                filter_expr = " and ".join(conditions)
   557→   557→        
   558→   558→        collection.load()
   559→   559→        
   560→   560→        search_params = {"metric_type": "COSINE", "params": {"nprobe": 10}}
   561→   561→        
   562→   562→        results = collection.search(
   563→   563→            data=[query_embedding],
   564→   564→            anns_field="embedding",
   565→   565→            param=search_params,
   566→   566→            limit=limit,
   567→   567→            expr=filter_expr,
   568→   568→            output_fields=["id", "agent_id", "type", "content", "importance", "created_at"]
   569→   569→        )
   570→   570→        
   571→   571→        memories = []
   572→   572→        for hits in results:
   573→   573→            for hit in hits:
   574→   574→                memories.append(self._result_to_memory(hit.entity))
   575→   575→        
   576→   576→        return memories
   577→   577→    
   578→   578→    async def delete(self, memory_id: UUID) -> bool:
   579→   579→        """Eliminar una memoria"""
   580→   580→        collection = await self._get_collection()
   581→   581→        
   582→   582→        collection.delete(f'id == "{str(memory_id)}"')
   583→   583→        collection.flush()
   584→   584→        
   585→   585→        return True
   586→   586→    
   587→   587→    async def update(self, memory_id: UUID, data: Dict[str, Any]) -> bool:
   588→   588→        """Actualizar una memoria (delete + insert)"""
   589→   589→        # Milvus no soporta update directo
   590→   590→        # Obtenemos, modificamos y reinsertamos
   591→   591→        memory = await self.retrieve(memory_id)
   592→   592→        if not memory:
   593→   593→            return False
   594→   594→        
   595→   595→        # Actualizar campos
   596→   596→        for key, value in data.items():
   597→   597→            if hasattr(memory, key):
   598→   598→                setattr(memory, key, value)
   599→   599→        
   600→   600→        memory.updated_at = datetime.utcnow()
   601→   601→        
   602→   602→        # Regenerar embedding si cambió el contenido
   603→   603→        if "content" in data and self.embedding_provider:
   604→   604→            memory.embedding = await self.embedding_provider.embed(memory.content)
   605→   605→        
   606→   606→        # Eliminar y reinsertar
   607→   607→        await self.delete(memory_id)
   608→   608→        await self.store(memory)
   609→   609→        
   610→   610→        return True
   611→   611→    
   612→   612→    def _result_to_memory(self, result: dict) -> Memory:
   613→   613→        """Convertir resultado de Milvus a modelo Memory"""
   614→   614→        return Memory(
   615→   615→            id=UUID(result.get("id")),
   616→   616→            agent_id=UUID(result.get("agent_id")),
   617→   617→            type=MemoryType(result.get("type")),
   618→   618→            content=result.get("content"),
   619→   619→            embedding=result.get("embedding"),
   620→   620→            importance=result.get("importance", 0.5),
   621→   621→            created_at=datetime.fromtimestamp(result.get("created_at", 0)),
   622→   622→            updated_at=datetime.utcnow(),
   623→   623→        )

   627→   627→# UNIFIED MEMORY SYSTEM

   629→   629→
   630→   630→class CognitiveMemorySystem:
   631→   631→    """
   632→   632→    Sistema de memoria cognitivo unificado.
   633→   633→    
   634→   634→    Coordina múltiples capas de memoria:
   635→   635→    - SQLite: Memoria de sesión y episódica
   636→   636→    - Milvus: Memoria semántica (vectorial)
   637→   637→    - Neo4j: Memoria procedural y declarativa (grafo)
   638→   638→    """
   639→   639→    
   640→   640→    def __init__(
   641→   641→        self,
   642→   642→        sqlite_store: Optional[SQLiteMemoryStore] = None,
   643→   643→        vector_store: Optional[VectorMemoryStore] = None,
   644→   644→        embedding_provider: Optional[EmbeddingProvider] = None
   645→   645→    ):
   646→   646→        self.session_store = sqlite_store or SQLiteMemoryStore()
   647→   647→        self.vector_store = vector_store
   648→   648→        self.embedding_provider = embedding_provider
   649→   649→    
   650→   650→    async def remember(
   651→   651→        self,
   652→   652→        agent_id: UUID,
   653→   653→        content: str,
   654→   654→        memory_type: MemoryType = MemoryType.SEMANTIC,
   655→   655→        session_id: Optional[UUID] = None,
   656→   656→        importance: float = 0.5,
   657→   657→        expires_in: Optional[int] = None
   658→   658→    ) -> Memory:
   659→   659→        """
   660→   660→        Almacenar un nuevo recuerdo.
   661→   661→        
   662→   662→        Args:
   663→   663→            agent_id: ID del agente que recuerda
   664→   664→            content: Contenido del recuerdo
   665→   665→            memory_type: Tipo de memoria
   666→   666→            session_id: ID de sesión (para memoria de sesión)
   667→   667→            importance: Importancia del recuerdo (0-1)
   668→   668→            expires_in: Tiempo de expiración en segundos
   669→   669→            
   670→   670→        Returns:
   671→   671→            La memoria creada
   672→   672→        """
   673→   673→        now = datetime.utcnow()
   674→   674→        expires_at = None
   675→   675→        if expires_in:
   676→   676→            expires_at = now + timedelta(seconds=expires_in)
   677→   677→        
   678→   678→        memory = Memory(
   679→   679→            id=uuid4(),
   680→   680→            agent_id=agent_id,
   681→   681→            type=memory_type,
   682→   682→            content=content,
   683→   683→            importance=importance,
   684→   684→            session_id=session_id,
   685→   685→            expires_at=expires_at,
   686→   686→            created_at=now,
   687→   687→            updated_at=now
   688→   688→        )
   689→   689→        
   690→   690→        # Almacenar según tipo
   691→   691→        if memory_type in [MemoryType.SESSION, MemoryType.EPISODIC]:
   692→   692→            await self.session_store.store(memory)
   693→   693→        elif memory_type == MemoryType.SEMANTIC:
   694→   694→            if self.vector_store:
   695→   695→                await self.vector_store.store(memory)
   696→   696→            else:
   697→   697→                # Fallback a SQLite si no hay vector store
   698→   698→                await self.session_store.store(memory)
   699→   699→        
   700→   700→        return memory
   701→   701→    
   702→   702→    async def recall(
   703→   703→        self,
   704→   704→        query: str,
   705→   705→        agent_id: Optional[UUID] = None,
   706→   706→        session_id: Optional[UUID] = None,
   707→   707→        limit: int = 10
   708→   708→    ) -> List[Memory]:
   709→   709→        """
   710→   710→        Recuperar recuerdos relevantes.
   711→   711→        
   712→   712→        Busca en todas las capas de memoria y combina resultados.
   713→   713→        """
   714→   714→        filters = {}
   715→   715→        if agent_id:
   716→   716→            filters["agent_id"] = agent_id
   717→   717→        if session_id:
   718→   718→            filters["session_id"] = session_id
   719→   719→        
   720→   720→        results = []
   721→   721→        
   722→   722→        # Buscar en memoria de sesión
   723→   723→        session_results = await self.session_store.search(query, limit, filters)
   724→   724→        results.extend(session_results)
   725→   725→        
   726→   726→        # Buscar en memoria semántica (vectorial)
   727→   727→        if self.vector_store:
   728→   728→            semantic_results = await self.vector_store.search(query, limit, filters)
   729→   729→            results.extend(semantic_results)
   730→   730→        
   731→   731→        # Deduplicar y ordenar por importancia
   732→   732→        seen_ids = set()
   733→   733→        unique_results = []
   734→   734→        for memory in results:
   735→   735→            if memory.id not in seen_ids:
   736→   736→                seen_ids.add(memory.id)
   737→   737→                unique_results.append(memory)
   738→   738→        
   739→   739→        unique_results.sort(key=lambda m: m.importance, reverse=True)
   740→   740→        
   741→   741→        return unique_results[:limit]
   742→   742→    
   743→   743→    async def forget(self, memory_id: UUID, memory_type: MemoryType) -> bool:
   744→   744→        """Olvidar (eliminar) un recuerdo"""
   745→   745→        if memory_type in [MemoryType.SESSION, MemoryType.EPISODIC]:
   746→   746→            return await self.session_store.delete(memory_id)
   747→   747→        elif memory_type == MemoryType.SEMANTIC and self.vector_store:
   748→   748→            return await self.vector_store.delete(memory_id)
   749→   749→        return False
   750→   750→    
   751→   751→    async def consolidate(
   752→   752→        self,
   753→   753→        session_id: UUID,
   754→   754→        agent_id: UUID
   755→   755→    ) -> List[Memory]:
   756→   756→        """
   757→   757→        Consolidar memorias de sesión en memoria semántica.
   758→   758→        
   759→   759→        Transfiere recuerdos importantes de la sesión actual
   760→   760→        a la memoria de largo plazo.
   761→   761→        """
   762→   762→        # Obtener memorias de sesión
   763→   763→        session_memories = await self.session_store.get_session_memories(session_id)
   764→   764→        
   765→   765→        consolidated = []
   766→   766→        
   767→   767→        for memory in session_memories:
   768→   768→            # Solo consolidar memorias importantes
   769→   769→            if memory.importance >= 0.7:
   770→   770→                # Crear memoria semántica
   771→   771→                semantic_memory = Memory(
   772→   772→                    id=uuid4(),
   773→   773→                    agent_id=agent_id,
   774→   774→                    type=MemoryType.SEMANTIC,
   775→   775→                    content=memory.content,
   776→   776→                    importance=memory.importance,
   777→   777→                    created_at=datetime.utcnow(),
   778→   778→                    updated_at=datetime.utcnow()
   779→   779→                )
   780→   780→                
   781→   781→                if self.vector_store:
   782→   782→                    await self.vector_store.store(semantic_memory)
   783→   783→                    consolidated.append(semantic_memory)
   784→   784→        
   785→   785→        return consolidated

   787→   787→    async def cleanup(self) -> Dict[str, int]:
   788→   788→        """Limpiar memorias expiradas"""
   789→   789→        deleted = await self.session_store.cleanup_expired()
   790→   790→        return {"deleted_count": deleted}
